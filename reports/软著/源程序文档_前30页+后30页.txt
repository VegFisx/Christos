"""
chemreact 核心：级联收敛
========================

把产物继续当底物往下反应，直到体系稳定。

设计要点（多数沿用了原 src/cascade_reactions.py 的正确思路）：

  1. ★ **每一轮重新计算条件激活**。
     原 cascade_reactions.py 是开头算一次就固定。但级联过程中体系在变
     （小分子被消耗、pH 可能漂移），条件层应该每轮重算。

  2. ★ **只迭代不可逆反应**。
     可逆反应很快达平衡，不会持续产生新分子。
     把可逆产物也丢进下一轮，级联会虚假膨胀、永不收敛。
     这一条原实现是对的，保留。

  3. ★ **副产物不进前沿**。
     水、甲醇、CO2 这类小分子不当作下一轮底物——
     它们要么是终产物，要么已经过量存在，再迭代没有意义。
     注意：是「不进前沿」，不是「从结果里删掉」，它们仍然出现在产物表中。

  4. ★ **每轮重算分数**。
     第 3 轮的产物和第 1 轮的产物，置信度不该一样——
     离原始成分越远，推断链条越长，不确定性越大。这里用轮数做衰减。

  5. **收敛判据**：一轮下来「新的不可逆产物」少于 min_new 就停。
"""

# 每远一轮，置信度乘这个系数。第 1 轮不衰减。
ROUND_DECAY = 0.85

# ★ 热力学覆盖率低于此值就整体回退启发式。
# 实测教训：中药大分子（附子里的乌头碱类，40~50 个重原子）超出 xTB 可靠范围，
# 覆盖率会低到 10%。这时如果坚持"能算的用真值、算不出的用启发式"，
# 同一张排行榜上就混了两套尺度，**比全部用启发式还糟**。
MIN_THERMO_COVERAGE = 0.5

# ★★ 传播分数门槛：只有分数够高的产物才继续往下反应 ★★
#
# 为什么需要它：甘草 15 个成分在煎煮条件下，8 轮涨到 1736 个产物且不见顶。
# 诊断发现，贡献最大的规则里一半 priority 只有 0.3（=在煎煮条件下只是猜测）——
# 把猜测级联下去，不确定性指数放大。
#
# ★ 为什么是「按分数」而不是「按规则优先级」——这是实测对比出来的：
#
#     策略                     轮数  收敛   产物总数  高分产物
#     A 现状（只按不可逆）        8    否     1736     595
#     B 按 priority >= 0.5       8    否     1378     561 ← 丢了 225 个高分产物
#     C 按 score >= 0.35         7    是     1680     595 ← 一个不丢
#     D 两者都加                 8    否     1392     561 ← 同样丢 225 个
#
#   按优先级设门槛会**截断通路**：高优先级规则固然可靠，但低优先级规则的产物
#   可能继续反应成高分产物，掐掉它们等于砍掉整条路径。
#   而分数里【已经含了 priority】（rule_match 权重 0.30）**又含了轮数衰减**，
#   是比 priority 单独更完整的判据——也是唯一能收敛且不丢高分产物的方案。
#
# 实测（甘草 15 个成分 / 煎煮 / max_rounds=8）：
#
#     门槛    轮数  收敛   产物总数   高分产物(>=0.5)
#     0.00     8    否      1510        573      ← 旧的"只按不可逆"
#     0.25     8    否      1522        573
#     0.35     7    是      1448        573      ← 默认值
#     0.45     5    是      1137        573      ← 想更快可以调到这个
#
#   机制：分数含轮数衰减 0.85^(r-1)，深处产物自然跌破门槛。
#   所以这个门槛本质上定义了「最多追几步推断」——
#   追得越深，累积不确定性越大，这个界是有化学意义的，不是拍脑袋。
MIN_PROPAGATE_SCORE = 0.35

def cascade(engine: Engine,
            seeds: Sequence[Compound],
            cond: Conditions,
            max_rounds: int = 3,
            threshold: float = 0.15,
            min_new: int = 10,
            frontier_limit: int = 300,
            thermo=None,
            thermo_top_k: int = 120,
            min_propagate_score: float = MIN_PROPAGATE_SCORE,
            ) -> CascadeResult:

    thermo:              core.thermo.Thermo 实例（可选）。
    thermo_top_k:        只给启发式排名前 k 的非副产物算 ΔG。
    min_propagate_score: ★传播门槛★ 只有分数 >= 此值的产物才继续往下反应。
                         这是让级联收敛的关键（见文件顶部常量处的实测对比）。
                         设成 0 就退回旧行为（不收敛）。

    round_stats: List[Dict[str, Any]] = []

    for r in range(1, max_rounds + 1):

            "new": len(new_this_round),

        products=rank(pool),

def _sample_frontier(products: Sequence[Product], limit: int,
                     next_round: int) -> List[Compound]:
    """按反应类型均衡采样，避免某一类规则（比如羟基氧化）淹没前沿。

    ★★ 必须确定性排序 ★★

    实测事故：把同一批成分的【输入顺序】打乱后重跑级联，
    输出只有 22~36% 自重叠（同一个引擎、同一批分子！）。

    根因：这里按【插入顺序】取前 per_rule 个，而插入顺序取决于
          输入成分的顺序 —— 于是"挑谁进前沿"变成了随机的。

    修法（两处都要确定）：
      1. 每条规则内部：按【分数降序、同分按 SMILES】排序后取前 per_rule 个
         （顺带带来好处：高置信产物优先进入前沿）
      2. 规则之间：轮转取（round-robin），规则按 id 排序
         —— 既保证每条规则都有机会，又不依赖 dict 的插入顺序

    修完之后：同一批成分不论以什么顺序传入，级联输出完全一致。
    """
    if len(products) <= limit:
        selected = list(products)
    else:
        by_rule: Dict[str, List[Product]] = defaultdict(list)
        for p in products:
            by_rule[p.rule_id].append(p)

        def _key(p: Product):
            return (-(p.score or 0.0), p.smiles)

        buckets = {rid: sorted(items, key=_key)
                   for rid, items in by_rule.items()}
        per_rule = max(1, limit // max(1, len(buckets)))

        # 轮转：第 0 轮每规则各取 1 个，第 1 轮再各取 1 个……
        selected = []
        depth = 0
        max_depth = max(len(v) for v in buckets.values())
        while len(selected) < limit and depth < max_depth:
            for rid in sorted(buckets):
                if depth < len(buckets[rid]):
                    selected.append(buckets[rid][depth])

------------------------------------------------------------
                    if len(selected) >= limit:
                        break
            depth += 1

    return [Compound(smiles=p.smiles, name=f"R{next_round} 中间体",
                     herb=p.source_herb, source="cascade")
            for p in selected]

def _fp(smiles: str, nbits: int):
    import numpy as np
    from rdkit import Chem, RDLogger
    from rdkit.Chem import rdFingerprintGenerator
    RDLogger.DisableLog("rdApp.*")
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=nbits)
    m = Chem.MolFromSmiles(smiles or "")
    if m is None:
        return np.zeros(nbits, dtype=np.float32)
    return np.array(gen.GetFingerprint(m), dtype=np.float32)

def shape_from_element_delta(element_delta) -> str:
    """★ D1（2026-10-13）**元素收支形状**：从 (dC, dH, dO) 判这一步的形状。

    口径写死（与 ORD 元素收支普查同一套形状定义）：
      * `dH <= -2 且 dO <= -1` -> `dehydration`（脱了一分子水 / 缩合形状）
      * `dH >= +2 且 dO >= +1` -> `hydrolysis`（加了一分子水 / 水解或加成形状）
      * 其它 -> 不作判断（返回空串）
    为什么它比 `shape_from_co_products` 更值得依赖：副产物是「同批其它产物」，与是否生成水无关；
    元素收支是**从反应式直接算出来的**，与模板出处无关（实测：9,770 条模板全是 2 个反应物模板，
    所以「两个底物」就是完整的反应物侧）。
    """
    try:
        dc, dh, do = (int(element_delta[0]), int(element_delta[1]), int(element_delta[2]))
    except Exception:                            # noqa: BLE001 —— 形状不对就不判，但要计数
        _SHAPE_FAILS["n"] += 1
        _SHAPE_FAILS["last"] = "element_delta 形状异常: %r" % (element_delta,)
        return ""
    del dc                                       # 只用水/氢的收支，碳数不参与判形状
    if dh <= -2 and do <= -1:
        return "dehydration"
    if dh >= 2 and do >= 1:
        return "hydrolysis"
    return ""

def _co_products_all_small(co_products, max_hac: int = 2) -> bool:
    """★ D1 收紧（2026-10-14，第四十三轮）：其余产物是否都是小分子（≤max_hac 重原子）。

    """
    for x in co_products or ():
        s = str(x or "").strip()
        if not s:
            continue
        try:
            from rdkit import Chem
            m = Chem.MolFromSmiles(s)
        except Exception as ex:                  # noqa: BLE001 —— 计数，不静默
            _SHAPE_FAILS["n"] += 1
            _SHAPE_FAILS["last"] = "%s: %s (%s)" % (type(ex).__name__, ex, s[:40])
            return False
        if m is None:

            _SHAPE_FAILS["last"] = "副产物解析失败: %s" % s[:40]
            return False
        if m.GetNumHeavyAtoms() > max_hac:
            return False
    return True

def classify_product(mol: Chem.Mol) -> Tuple[bool, bool]:
    """返回 (是否保留, 是否小分子副产物)。

    这是对原 _valid_product 的替代。**只丢真正不是分子的东西**，
    小分子一律保留但打标记。
    """
    try:
        if Chem.SanitizeMol(Chem.Mol(mol), catchErrors=True) != Chem.SanitizeFlags.SANITIZE_NONE:
            return False, False
    except Exception:
        return False, False

    mw = Descriptors.MolWt(mol)
    heavy = mol.GetNumHeavyAtoms()

    if mw < DISCARD_MIN_MW or heavy < 1:
        return False, False          # H+、单原子离子

    is_byproduct = (mw <= BYPRODUCT_MAX_MW) or (heavy <= BYPRODUCT_MAX_HEAVY)
    return True, is_byproduct

def _counts_differ(a: Dict[str, float], b: Dict[str, float],
                   tol: float = 1e-6) -> bool:
    """两边原子数是否不同。任一边为空（解析失败）时返回 False（保守放行）。"""
    if not a or not b:
        return False
    for k in set(a) | set(b):
        if abs(a.get(k, 0.0) - b.get(k, 0.0)) > tol:
            return True
    return False

def predict_reactions(compounds: List[Dict],
                      conditions: Optional[Conditions] = None,
                      min_mw: float = 100.0,
                      min_heavy: int = 10,
                      gate_conditions: bool = True) -> List[Dict]:
    """与 TCMAI `predict_reactions()` 同形状：见文件顶部接口说明。

    额外多给三个键（不会影响旧调用方）：`activation`（条件门控分数）、
    `category`、`gate_note`（被降权/判零的原因）。
    """
    cond = conditions or DEFAULT_CONDITIONS
    eng = get_engine(gate_conditions=gate_conditions)
    comps: List[CRCompound] = []
    meta: Dict[str, Dict[str, str]] = {}
    for c in compounds:
        if not isinstance(c, dict):
            continue
        smi = c.get("smiles")
        if not smi:
            continue
        name = c.get("name", "(无名)")
        herb = c.get("herb_name") or c.get("source", "")
        comps.append(CRCompound(smiles=smi, name=name, herb=herb, source="herb"))
        meta[CRCompound(smiles=smi, name=name, herb=herb, source="herb").to_dict()["smiles"]] = {
            "name": name, "herb": herb}
    if len(comps) < 2:
        return []

    prods = eng.step(comps, cond, round_no=1)
    out, seen = [], set()
    for p in prods:
        if p.mw < min_mw or p.heavy_atoms < min_heavy:
            continue
        key = (p.smiles, p.source_name)
        if key in seen:
            continue
        seen.add(key)
        out.append({
            "from_herb": p.source_herb,
            "from_compound": p.source_name,
            "rxn": p.rule_name,
            "formula": p.formula,
            "smiles": p.smiles,
            "mw": round(p.mw, 1),
            # —— 扩展字段（本桥新增）——
            "activation": round(float(getattr(p, "activation", 0.0)), 3),
            "category": getattr(p, "category", ""),
            "gate_note": "",

        })
    return out


def predict_for_herb(herb_name: str, conditions: Optional[Conditions] = None,
                     limit: int = 200) -> Dict[str, Any]:
    """只读演示：从 tcm.db 取药材成分 -> 预测产物。"""
    from tcm_ro import use_readonly; use_readonly(); from db_utils import find_compounds  # 来自 D:\TCMAI\src（只读）
    rows = find_compounds(herb_name, limit=limit)
    prods = predict_reactions(rows, conditions)
    types: Dict[str, int] = {}
    for p in prods:
        types[p["rxn"]] = types.get(p["rxn"], 0) + 1
    return {"herb": herb_name, "compounds": len(rows),
            "cond": (conditions or DEFAULT_CONDITIONS).describe(),
            "n_products": len(prods), "types": dict(sorted(types.items(), key=lambda kv: -kv[1])),
            "products": prods}
